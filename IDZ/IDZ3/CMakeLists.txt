cmake_minimum_required(VERSION 3.27)
project(idz3 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Open this project in Ubuntu/WSL.")
endif()
find_package(Threads REQUIRED)
add_library(garden3_core STATIC src/model.c src/ipc.c src/worker.c)
target_compile_features(garden3_core PUBLIC c_std_17)
target_compile_definitions(garden3_core PUBLIC _POSIX_C_SOURCE=200809L)
target_compile_options(garden3_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(garden3_core PUBLIC Threads::Threads rt)
foreach(program gardeners_named gardeners_unnamed garden_server gardener)
    if(program STREQUAL "gardener")
        add_executable(${program} src/worker_main.c)
    else()
        add_executable(${program} src/main.c)
    endif()
    target_compile_options(${program} PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(${program} PRIVATE garden3_core)
endforeach()
target_compile_definitions(gardeners_named PRIVATE FORK_MODE=FORK_NAMED)
target_compile_definitions(gardeners_unnamed PRIVATE FORK_MODE=FORK_UNNAMED)
include(CTest)
if(BUILD_TESTING)
    foreach(mode named unnamed shm mq)
        foreach(scenario demo rows blocked isolated waiting overflow interrupt)
            add_test(NAME idz3.${mode}.${scenario} COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.sh"
                "${CMAKE_CURRENT_BINARY_DIR}" ${mode} ${scenario})
            set_tests_properties(idz3.${mode}.${scenario} PROPERTIES TIMEOUT 20)
        endforeach()
    endforeach()
endif()
