cmake_minimum_required(VERSION 3.27)
project(idz2 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Open this project in Ubuntu/WSL.")
endif()
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)

# Две программы используют общую модель, отличается только ожидание изменений.
foreach(mode sem cond)
    set(target gardeners2_${mode})
    add_executable(${target} src/main.c src/garden.c src/gardener.c)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(mode STREQUAL "cond")
        target_compile_definitions(${target} PRIVATE USE_CONDVAR)
        set_target_properties(${target} PROPERTIES OUTPUT_NAME gardeners_cond)
    else()
        set_target_properties(${target} PROPERTIES OUTPUT_NAME gardeners)
    endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
    foreach(mode sem cond)
        foreach(scenario demo rows blocked isolated two_cells waiting invalid overflow interrupt)
            add_test(NAME idz2.${mode}.${scenario} COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.sh"
                "$<TARGET_FILE:gardeners2_${mode}>" ${scenario})
            set_tests_properties(idz2.${mode}.${scenario} PROPERTIES TIMEOUT 15)
        endforeach()
    endforeach()
endif()
