cmake_minimum_required(VERSION 3.27)
project(idz4 LANGUAGES C)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Open this project in Linux or Ubuntu/WSL.")
endif()

add_library(garden4_common STATIC src/model.c src/net.c)
target_compile_definitions(garden4_common PUBLIC _POSIX_C_SOURCE=200809L)
add_executable(garden4_server src/server.c)
add_executable(garden4_worker src/client.c)
add_executable(garden4_view src/client.c)
target_compile_definitions(garden4_view PRIVATE OBSERVER)
foreach(target garden4_common garden4_server garden4_worker garden4_view)
    set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
foreach(target garden4_server garden4_worker garden4_view)
    target_link_libraries(${target} PRIVATE garden4_common)
endforeach()

include(CTest)
if(BUILD_TESTING)
    foreach(scenario basic demo rows isolated blocked yield corridor yield_reconnect viewers reconnect worker_move shutdown invalid)
        add_test(NAME idz4.${scenario}
            COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.sh" "${CMAKE_CURRENT_BINARY_DIR}" ${scenario})
        set_tests_properties(idz4.${scenario} PROPERTIES TIMEOUT 20)
    endforeach()
endif()
